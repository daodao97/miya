package main

import (
	"os"
	"path/filepath"
	"strings"
	"testing"
)

// These check serialization through NSGo's plist encoder and the actual
// profile prefix/grant rules, where a wrong package can build but fail to install.
func TestMiyaPackagingPlist(t *testing.T) {
	p, err := load(os.Getenv("MIYA_PACKAGING_ROOT"))
	if err != nil {
		t.Fatal(err)
	}
	p.MiyaTarget = "macos_arm64"
	path := filepath.Join(t.TempDir(), "Info.plist")
	if err := writePlist(path, p, macPlist, ""); err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	value, err := decodePlist(data)
	if err != nil {
		t.Fatal(err)
	}
	info := value.(map[string]any)
	if info["LSMinimumSystemVersion"] != "13.0" || info["CFBundleVersion"] != p.Version {
		t.Fatalf("desktop metadata: %v", info)
	}
	types := info["CFBundleURLTypes"].([]any)
	schemes := types[0].(map[string]any)["CFBundleURLSchemes"].([]any)
	if len(schemes) != 1 || schemes[0] != "miya" {
		t.Fatalf("schemes: %v", schemes)
	}
	langs := info["CFBundleLocalizations"].([]any)
	if len(langs) != 2 || langs[1] != "zh-Hans" {
		t.Fatalf("localizations: %v", langs)
	}
	if !strings.Contains(p.ldflags(), "miya-release:"+p.Version+":end") {
		t.Fatal("release linker marker missing")
	}
}

func TestMiyaPackagingKeychainProfile(t *testing.T) {
	p, err := load(os.Getenv("MIYA_PACKAGING_ROOT"))
	if err != nil {
		t.Fatal(err)
	}
	makeProfile := func(groups ...any) map[string]any {
		return map[string]any{"application-identifier": "OLDPREFIX.com.daodao.miya", "keychain-access-groups": groups}
	}
	ents := makeProfile("OLDPREFIX.*", "com.apple.token")
	if err := miyaIOSEntitlements(p, ents, true); err != nil {
		t.Fatal(err)
	}
	groups := ents["keychain-access-groups"].([]any)
	if groups[0] != "OLDPREFIX.*" || groups[1] != "com.apple.token" {
		t.Fatalf("profile prefix was replaced by team id: %v", groups)
	}
	if err := miyaIOSEntitlements(p, makeProfile("OLDPREFIX.*"), true); err == nil {
		t.Fatal("signed without a grant for the legacy com.apple.token group")
	}
	if err := miyaIOSEntitlements(p, makeProfile("NEWPREFIX.*", "com.apple.token"), true); err == nil {
		t.Fatal("signed with a Keychain prefix not granted by the profile")
	}
}
