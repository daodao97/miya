package apk

import (
	"encoding/xml"
	"fmt"
	"strings"
)

type IntentQuery struct {
	Action   string `yaml:"action"`
	MIMEType string `yaml:"mime_type"`
}

func miyaNetworkID(m Manifest) uint32 {
	id := uint32(1)
	if m.Icon != nil {
		id++
	}
	if len(m.Names) > 0 {
		id++
	}
	return appPackageID<<24 | id<<16
}

func miyaManifest(root, app *node, m Manifest) {
	for k, v := range m.AppAttributes {
		setAttr(app, androidNS, k, fmt.Sprint(v))
	}
	activity := child(app, "activity")
	if activity != nil {
		for k, v := range m.ActivityAttributes {
			setAttr(activity, androidNS, k, fmt.Sprint(v))
		}
		for _, a := range activity.attrs {
			if a.Name.Local == "configChanges" && len(m.ExtraConfigChanges) > 0 {
				setAttr(activity, androidNS, "configChanges", a.Value+"|"+strings.Join(m.ExtraConfigChanges, "|"))
			}
		}
	}
	for _, feature := range m.OptionalFeatures {
		n := &node{name: xml.Name{Local: "uses-feature"}}
		setAttr(n, androidNS, "name", feature)
		setAttr(n, androidNS, "required", "false")
		root.kids = append(root.kids, n)
	}
	if len(m.Queries) > 0 {
		queries := &node{name: xml.Name{Local: "queries"}}
		for _, q := range m.Queries {
			intent := &node{name: xml.Name{Local: "intent"}}
			action := &node{name: xml.Name{Local: "action"}}
			setAttr(action, androidNS, "name", q.Action)
			intent.kids = append(intent.kids, action)
			if q.MIMEType != "" {
				data := &node{name: xml.Name{Local: "data"}}
				setAttr(data, androidNS, "mimeType", q.MIMEType)
				intent.kids = append(intent.kids, data)
			}
			queries.kids = append(queries.kids, intent)
		}
		root.kids = append(root.kids, queries)
	}
	if len(m.NetworkSecurityXML) > 0 {
		setAttr(app, androidNS, "networkSecurityConfig", "@xml/miya_network_security")
	}
}

func miyaAPKResources(pkg string, icon bool, names map[string]string, network bool) []byte {
	return table(pkg, miyaResourceTypes(icon, names, network))
}
func miyaNetworkXML(data []byte) ([]byte, error) {
	root, err := parseXML(data)
	if err != nil {
		return nil, err
	}
	return encodeMiyaXML(root, Manifest{})
}
func parseMiyaLaunchMode(s string) (uint32, error) {
	if s == "singleTask" {
		return 2, nil
	}
	return 0, fmt.Errorf("unsupported launchMode %q", s)
}
